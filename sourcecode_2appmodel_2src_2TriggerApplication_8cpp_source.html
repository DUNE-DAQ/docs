83std::vector<const confmodel::DaqModule*>
84TriggerApplication::generate_modules(conffwk::Configuration* confdb,
85 const std::string& dbfile,
86 const confmodel::Session* session) const
87{
88 std::vector<const confmodel::DaqModule*> modules;
89
90 auto ti_conf = get_trigger_inputs_handler();
91 auto ti_class = ti_conf->get_template_for();
92 std::string handler_name("");
93 // Process the queue rules looking for inputs to our trigger handler modules
94 const QueueDescriptor* ti_inputq_desc = nullptr;
95
96 for (auto rule : get_queue_rules()) {
97 auto destination_class = rule->get_destination_class();
98 auto data_type = rule->get_descriptor()->get_data_type();
99 if (destination_class == "DataHandlerModule" || destination_class == ti_class) {
100 ti_inputq_desc = rule->get_descriptor();
101 }
102 }
103 // Process the network rules looking for the TP handler data reuest inputs
104 const NetworkConnectionDescriptor* req_net_desc = nullptr;
105 const NetworkConnectionDescriptor* tin_net_desc = nullptr;
106 const NetworkConnectionDescriptor* tout_net_desc = nullptr;
107 const NetworkConnectionDescriptor* tset_out_net_desc = nullptr;
108 for (auto rule : get_network_rules()) {
109 auto endpoint_class = rule->get_endpoint_class();
110 auto data_type = rule->get_descriptor()->get_data_type();
111
112 if (data_type == "DataRequest") {
113 req_net_desc = rule->get_descriptor();
114 }
115 else if (data_type == "TASet" || data_type == "TCSet"){
116 tset_out_net_desc = rule->get_descriptor();
117 }
118 else if (endpoint_class == "DataSubscriberModule") {
119 if (!tin_net_desc) {
120 tin_net_desc = rule->get_descriptor();
121 }
122 else if (rule->get_descriptor()->get_data_type() == tin_net_desc->get_data_type()) {
123 // For now endpoint_class of DataSubscriberModule for both input and output
124 // with the same data type is not possible.
125 throw (BadConf(ERS_HERE, "Have two network connections of the same data_type and the same endpoint_class"));
126 }
127 else if (tin_net_desc->get_data_type() == "TriggerActivity" &&
128 rule->get_descriptor()->get_data_type() == "TriggerCandidate") {
129 // For TA->TC
130 tout_net_desc = rule->get_descriptor();
131 handler_name = "tahandler";
132 }
133 else if (tin_net_desc->get_data_type() == "TriggerCandidate" &&
134 rule->get_descriptor()->get_data_type() == "TriggerActivity") {
135 // For TA->TC if we saved TC network connection as input first...
136 tout_net_desc = tin_net_desc;
137 tin_net_desc = rule->get_descriptor();
138 handler_name = "tahandler";
139 }
140 else {
141 throw (BadConf(ERS_HERE, "Unexpected input & output network connection descriptors provided"));
142 }
143 }
144 else if (data_type == "TriggerActivity" || data_type == "TriggerCandidate"){
145 tout_net_desc = rule->get_descriptor();
146 if (data_type == "TriggerActivity")
147 handler_name = "tphandler";
148 else
149 handler_name = "tahandler";
150 }
151 }
152
153 // Process special Network rules!
154 // Looking for Fragment rules from DFAppplications in current Session
155 auto sessionApps = session->get_enabled_applications();
156 std::vector<conffwk::ConfigObject> fragOutObjs;
157 for (auto app : sessionApps) {
158 auto dfapp = app->cast<appmodel::DFApplication>();
159 if (dfapp == nullptr)
160 continue;
161
162 auto dfNRules = dfapp->get_network_rules();
163 for (auto rule : dfNRules) {
164 auto descriptor = rule->get_descriptor();
165 auto data_type = descriptor->get_data_type();
166 if (data_type == "Fragment") {
167 std::string dreqNetUid(descriptor->get_uid_base() + dfapp->UID());
168 conffwk::ConfigObject frag_conn;
169 confdb->create(dbfile, "NetworkConnection", dreqNetUid, frag_conn);
170
171 frag_conn.set_by_val<std::string>("data_type", descriptor->get_data_type());
172 frag_conn.set_by_val<std::string>("connection_type", descriptor->get_connection_type());
173
174 auto serviceObj = descriptor->get_associated_service()->config_object();
175 frag_conn.set_obj("associated_service", &serviceObj);
176 fragOutObjs.push_back(frag_conn);
177 } // If network rule has TriggerDecision type of data
178 } // Loop over Apps network rules
179 } // loop over Session specific Apps
180
181 // Now create the Data Handler and its associated queue and network
182 // connections
183 conffwk::ConfigObject input_queue_obj;
184 conffwk::ConfigObject req_net_obj;
185 conffwk::ConfigObject tin_net_obj;
186 conffwk::ConfigObject tout_net_obj;
187 conffwk::ConfigObject tset_out_net_obj;
188
189 if ( req_net_desc== nullptr) {
190 throw (BadConf(ERS_HERE, "No network descriptor given to receive request and send data was set"));
191 }
192 if ( tin_net_desc== nullptr) {
193 throw (BadConf(ERS_HERE, "No network descriptor given to receive trigger objects"));
194 }
195 if ( tout_net_desc== nullptr) {
196 throw (BadConf(ERS_HERE, "No network descriptor given to publish trigger objects"));
197 }
198 if (ti_inputq_desc == nullptr) {
199 throw (BadConf(ERS_HERE, "No data input queue descriptor given"));
200 }
201
202 std::string queue_uid(ti_inputq_desc->get_uid_base());
203 confdb->create(dbfile, "Queue", queue_uid, input_queue_obj);
204 input_queue_obj.set_by_val<std::string>("data_type", ti_inputq_desc->get_data_type());
205 input_queue_obj.set_by_val<std::string>("queue_type", ti_inputq_desc->get_queue_type());
206 input_queue_obj.set_by_val<uint32_t>("capacity", ti_inputq_desc->get_capacity());
207
208
209 req_net_obj = create_network_connection(req_net_desc->get_uid_base()+UID(),
210 req_net_desc, confdb, dbfile);
211
212 tin_net_obj = create_network_connection(tin_net_desc->get_uid_base()+".*",
213 tin_net_desc, confdb, dbfile);
214
215 tout_net_obj = create_network_connection(tout_net_desc->get_uid_base()+UID(),
216 tout_net_desc, confdb, dbfile);
217
218 if (tset_out_net_desc) {
219 tset_out_net_obj = create_network_connection(tset_out_net_desc->get_uid_base()+UID(),
220 tset_out_net_desc, confdb, dbfile);
221 }
222
223 // build up the full list of outputs
224 std::vector<const conffwk::ConfigObject*> ti_output_objs;
225 for (auto& fNet : fragOutObjs) {
226 ti_output_objs.push_back(&fNet);
227 }
228 ti_output_objs.push_back(&tout_net_obj);
229 if (tset_out_net_desc!= nullptr) {
230 ti_output_objs.push_back(&tset_out_net_obj);
231 }
232
233 auto ti_conf_obj = ti_conf->config_object();
234 conffwk::ConfigObject ti_obj;
235 if (get_source_id() == nullptr) {
236 throw(BadConf(ERS_HERE, "No source_id associated with this TriggerApplication!"));
237 }
238 uint32_t source_id = get_source_id()->get_sid();
239 std::string ti_uid(handler_name + "-" + std::to_string(source_id));
240 confdb->create(dbfile, ti_class, ti_uid, ti_obj);
241 ti_obj.set_by_val<uint32_t>("source_id", source_id);
242 ti_obj.set_by_val<uint32_t>("detector_id", 1); // 1 == kDAQ
243 ti_obj.set_by_val<bool>("post_processing_enabled", !get_tx_generation_disabled());
244
245 ti_obj.set_obj("module_configuration", &ti_conf_obj);
246 ti_obj.set_objs("inputs", {&input_queue_obj, &req_net_obj});
247 ti_obj.set_objs("outputs", ti_output_objs);
248 // Add to our list of modules to return
249 modules.push_back(confdb->get<DataHandlerModule>(ti_uid));
250
251
252 // Now create the DataSubscriberModule object
253 auto rdr_conf = get_data_subscriber();
254 if (rdr_conf == nullptr) {
255 throw (BadConf(ERS_HERE, "No DataReaderModule configuration given"));
256 }
257
258 // Create a DataReaderModule
259
260 std::string reader_uid("data-reader-"+UID());
261 std::string reader_class = rdr_conf->get_template_for();
262 conffwk::ConfigObject reader_obj;
263 TLOG_DEBUG(7) << "creating OKS configuration object for Data subscriber class " << reader_class;
264 confdb->create(dbfile, reader_class, reader_uid, reader_obj);
265 reader_obj.set_objs("inputs", {&tin_net_obj} );
266 reader_obj.set_objs("outputs", {&input_queue_obj} );
267 reader_obj.set_obj("configuration", &rdr_conf->config_object());
268
269 modules.push_back(confdb->get<DataSubscriberModule>(reader_uid));
270
271 return modules;
272}
