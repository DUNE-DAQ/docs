161int dbse::SchemaMainWindow::ShouldSaveViewChanges() const

163 QString modified_views;
164 for (int index=0; index<ui->TabWidget->count(); ++index) {
165 auto tab = dynamic_cast<SchemaTab *> (ui->TabWidget->widget(index));
166 if (tab->GetScene()->IsModified()) {

169 }

171 QString message ( "There are unsaved changes in the schema views:\n");
172 message.append (modified_views);
173 message.append ("Do you want to save the changes in the schema views?\n" );
174 return QMessageBox::question (
175 0, tr ( "SchemaEditor" ),
176 message,
177 QMessageBox::Discard | QMessageBox::Cancel, QMessageBox::Discard );
178 }

206void dbse::SchemaMainWindow::RemoveClass()
207{
208 QModelIndex Index = ui->ClassTableView->currentIndex();
209 QModelIndex proxyIndex = proxyModel->mapToSource( Index );
210 QStringList Row = TableModel->getRowFromIndex ( proxyIndex );
211 OksClass * SchemaClass = KernelWrapper::GetInstance().FindClass ( Row.at (
212 0 ).toStdString() );
213
214 if ( SchemaClass->all_sub_classes()->size() != 0 )
215 {
216 QMessageBox::warning ( 0, "Schema editor",
217 QString ( "Cannot delete class because it has sub-classes." ) );
218 return;
219 }
220 else if ( KernelWrapper::GetInstance().AnyClassReferenceThis ( SchemaClass ) )
221 {
222 QMessageBox::warning ( 0, "Schema editor",
223 QString ( "Cannot delete class because some classes references it." ) );
224 return;
225 }
226 else
227 {
228 QString Name = QString::fromStdString ( SchemaClass->get_name() );
229 KernelWrapper::GetInstance().PushRemoveClassCommand ( SchemaClass, SchemaClass->get_name(),
230 SchemaClass->get_description(),
231 SchemaClass->get_is_abstract() );
232 BuildFileModel();
233 }

510void dbse::SchemaMainWindow::SaveView()
511{
512 SchemaTab * CurrentTab = dynamic_cast<SchemaTab *> ( ui->TabWidget->currentWidget() );
513
514 if ( CurrentTab->GetScene()->items().size() != 0 )
515 {
516 auto defName = CurrentTab->getFileName();
517 if (defName == "./") {
518 defName = m_view_dir;
519 }
520 QString FileName = QFileDialog::getSaveFileName (
521 this, tr ( "Save View" ), defName );
522
523 if (! FileName.isEmpty()) {
524 auto spos = FileName.lastIndexOf('/');
525 if (spos != -1) {
526 m_view_dir = FileName;
527 m_view_dir.truncate(spos);
528 }
529
530 if ( !FileName.endsWith ( ".view" ) ) {
531 FileName.append ( ".view" );
532 }
533
534 CurrentTab->setFileName ( FileName );
535 if (CurrentTab->getName() == "") {
536 auto text = QFileInfo(FileName).baseName();
537 CurrentTab->setName(text);
538 auto index = ui->TabWidget->currentIndex();
539 ui->TabWidget->setTabText(index, text);
540 }
541
542 QFile ViewFile ( FileName );
543 ViewFile.open ( QIODevice::WriteOnly );
544
545 for ( QGraphicsItem * Item : CurrentTab->GetScene()->items() ) {
546 if ( dynamic_cast<SchemaGraphicObject *> ( Item ) ) {
547 SchemaGraphicObject * SchemaObject = dynamic_cast<SchemaGraphicObject *> ( Item );
548 QString ObjectDescription = SchemaObject->GetClassName() + ","
549 + QString::number ( SchemaObject->scenePos().x() ) + ","
550 + QString::number ( SchemaObject->scenePos().y() ) + "\n";
551 ViewFile.write ( ObjectDescription.toUtf8() );
552 }
553 }
554
555 ViewFile.close();
556 auto message = QString("Saved view to %1").arg(FileName);
557 ui->StatusBar->showMessage( message );
558 CurrentTab->GetScene()->ClearModified();
559 }
560 }

563void dbse::SchemaMainWindow::LoadView()
564{
565 QString ViewPath = QFileDialog::getOpenFileName (
566 this,
567 tr ("Open view file"),
568 m_view_dir,
569 "*.view");
570
571 if ( !ViewPath.isEmpty() )
572 {
573 auto spos = ViewPath.lastIndexOf('/');
574 if (spos != -1) {
575 m_view_dir = ViewPath;
576 m_view_dir.truncate(spos);
577 }
578 QFile ViewFile ( ViewPath );
579 ViewFile.open ( QIODevice::ReadOnly );
580
581 auto text = QFileInfo(ViewPath).baseName();
582 auto index = ui->TabWidget->addTab ( new SchemaTab(),
583 text );
584 ui->TabWidget->setCurrentIndex ( index );
585 SchemaTab * CurrentTab = dynamic_cast<SchemaTab *> ( ui->TabWidget->currentWidget() );
586 CurrentTab->setName(text);
587 CurrentTab->setFileName(ViewPath);
588
589 QStringList ClassesNames;
590 QList<QPointF> Positions;
591
592 while ( !ViewFile.atEnd() )
593 {
594 QString Line ( ViewFile.readLine() );
595 QStringList ObjectDescription = Line.split ( "," );
596 ClassesNames.append ( ObjectDescription.at ( 0 ) );
597 QPointF Position;
598 Position.setX ( ObjectDescription.at ( 1 ).toInt() );
599 Position.setY ( ObjectDescription.at ( 2 ).toInt() );
600 Positions.append ( Position );
601 }
602 ViewFile.close();
603
604 auto message = QString("Loaded view from %1").arg(ViewPath);
605 ui->StatusBar->showMessage( message );
606
607 auto scene = CurrentTab->GetScene();
608 scene->CleanItemMap();
609 auto missing = scene->AddItemsToScene ( ClassesNames, Positions );
610 if (!missing.empty()) {
611 QString text{"The following classes in "};
612 text.append(QFileInfo(ViewPath).fileName());
613 text.append(" are not present in the loaded schema:\n ");
614 text.append(missing.join(",\n "));
615 QMessageBox::warning(this, tr("Load View"), text);
616 }
617 scene->ClearModified();
618 }
