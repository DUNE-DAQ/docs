294uint16_t
295DaphneV2ADC::get_reg4() const {
296
297 // ADC, reg 4 has no parsing as it's all made of booleans
298 std::bitset<5> reg4;
299 // bits 0 and 2 are reserved
300 reg4[1] = get_low_resolution();
301 reg4[3] = get_output_offset_binary();
302 reg4[4] = get_MSB_first();
303 return reg4.to_ulong();
304}

306uint16_t
307DaphneV2PGA::get_reg51() const {
308
309 std::bitset<14> reg51(get_lpf_cut_frequency());
310 reg51 <<= 1;
311 reg51[4] = get_integrator_disable();
312 reg51[7] = true; // clamp is always disabled and we are in low noise mode
313 reg51[13] = get_gain();
314
315 return reg51.to_ulong() ;
316}

318uint16_t
319DaphneV2LNA::get_reg52() const {
320
321 std::bitset<16> reg52;
322
323 decltype(reg52) clamp(get_clamp());
324 clamp <<= 6;
325
326 reg52[12] = get_integrator_disable();
327
328 decltype(reg52) gain(get_gain());
329 clamp <<= 13;
330
331 reg52 |= clamp;
332 reg52 |= gain;
333
334 return reg52.to_ulong();
335}
