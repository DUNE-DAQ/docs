40SSPLEDCalibWrapper::init(const appmodel::SSPLEDCalibModule* conf)
41{
42 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "SSPLEDCalibWrapper::init called." << std::endl;
43
44 m_device_interface = new dunedaq::sspmodules::DeviceInterface();
45
46 m_number_channels = conf->get_number_channels();
47 m_channel_mask = conf->get_channel_mask();
48 m_burst_count = conf->get_burst_count();
49 m_double_pulse_delay_ticks = conf->get_double_pulse_delay_ticks();
50 m_pulse1_width_ticks = conf->get_pulse1_width_ticks();
51 m_pulse2_width_ticks = conf->get_pulse2_width_ticks();
52 m_pulse_bias_percent_270nm = conf->get_pulse_bias_percent_270nm();
53 m_pulse_bias_percent_367nm = conf->get_pulse_bias_percent_367nm();
54
55 m_board_id = conf->get_board_id();
56 m_module_id = conf->get_module_id();
57 m_instance_name_for_metrics = "SSP LED Calib " + std::to_string(m_board_id);
58 m_partition_number =conf->get_partition_number(); // this should be 0-3
59
60 m_timing_address = conf->get_timing_address(); // 0x20 is default for 101, 0x2B for 304, and 0x36 for 603
61 if (m_timing_address > 0xff) {
62 std::stringstream ss;
63 ss << "Error: Incorrect timing address set (" << m_timing_address << ")!" << std::endl;
64 TLOG() << ss.str();
65 throw ConfigurationError(ERS_HERE, ss.str());
66 }
67
68 TLOG_DEBUG(TLVL_WORK_STEPS) << "Board ID is listed as: " << m_board_id << std::endl
69 << "Partition Number is: " << m_partition_number << std::endl
70 << "Timing Address is: " << m_timing_address << std::endl
71 << "Module ID is: " << m_module_id << std::endl;
72
73 try {
74 m_device_interface->SetPartitionNumber(m_partition_number);
75 m_device_interface->SetTimingAddress(m_timing_address);
76 m_device_interface->ConfigureLEDCalib(conf); //This sets up the ethernet interface and make sure that the pdts is synched
77 m_device_interface->SetRegisterByName("module_id", m_module_id);
78 //m_device_interface->SetRegisterByName("eventDataInterfaceSelect", m_cfg.interface_type);
79 } catch (const std::exception & e) {
80 throw FailedLEDCalibration(ERS_HERE, e);
81 }
82
83 if ( conf->get_pulse_mode() == "single") {
84 m_single_pulse = true;
85 TLOG(TLVL_FULL_DEBUG) << "SSPLEDCalibWrapper: I think that you want SSP LED Calib module to be in single pulse..." << std::endl;
86 } else if ( conf->get_pulse_mode() == "burst") {
87 m_burst_mode = true;
88 TLOG(TLVL_FULL_DEBUG) << "SSPLEDCalibWrapper: I think that you want SSP LED Calib module to be in BURST MODE..." << std::endl;
89 }
90
91 if ( (m_single_pulse && m_burst_mode) ) {
92 std::stringstream ss;
93 ss << "ERROR: SOMEHOW ENDED UP WITH SEVERAL PULSE MODES SET ON!!!!" << std::endl;
94 TLOG() << ss.str();
95 throw ConfigurationError(ERS_HERE, ss.str());
96 }
97
98 if (m_burst_mode) {
99 TLOG(TLVL_FULL_DEBUG) << "SSPLEDCalibWrapper: Configuring for BURST MODE..." << std::endl;
100 this->configure_burst_mode();
101 } else if (m_single_pulse) {
102 TLOG(TLVL_FULL_DEBUG) << "SSPLEDCalibWrapper: Configuring for single pulse mode..." << std::endl;
103 this->configure_single_pulse();

105 std::stringstream ss;
106 ss << "ERROR: SOMEHOW ENDED UP WITH NO PULSE MODES SET ON!!!!" << std::endl;
107 TLOG() << ss.str();
108 throw ConfigurationError(ERS_HERE, ss.str());
109 }
110
111 //if there are "literal" entries in the configuration they are explicit writes to the specified register with given value
112 //these literal entries are paresed and applied last after any other parameters so this method call needs to be after the
113 //other configuration calls
114 this->manual_configure_device(conf->get_hardware_configuration());
115
116 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "SSPLEDCalibWrapper::configure complete.";

119void
120SSPLEDCalibWrapper::start(const data_t& /*args*/)
121{
122 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "Start pulsing SSPLEDCalibWrapper of card " << m_board_id << "...";
123
124 if (m_run_marker.load()) {
125 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "Run Marker says that SSPLEDCalibWrapper card " << m_board_id << " is already pulsing...";
126 return;
127 }
128
129 unsigned int base_bias_regAddress = 0x40000340;
130 unsigned int base_timing_regAddress = 0x800003C0;
131
132 if (m_number_channels == 5) {
133 base_bias_regAddress = 0x4000035C;
134 base_timing_regAddress = 0x800003DC;
135 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "Configuring board with 5 channels..." ;
136 } else if (m_number_channels == 12) {
137 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "Configuring board with 12 channels..." ;
138 } else {
139 std::stringstream ss;
140 ss << "ERROR: SOMEHOW ENDED UP WITHOUT NUMBER OF CHANNELS SET!!!!" << std::endl;
141 TLOG() << ss.str();
142 throw ConfigurationError(ERS_HERE, ss.str());
143 }
144
145 unsigned int pulse_bias_setting_270nm = ( m_pulse_bias_percent_270nm);
146 unsigned int pulse_bias_setting_367nm = ( m_pulse_bias_percent_367nm/2);
147
148 for (unsigned int counter = 0; counter < m_number_channels ; counter++) {
149 unsigned int bias_regAddress = base_bias_regAddress + 0x4*(counter); //0x40000340 - 0x4000036C
150 unsigned int bias_regVal = 0x00040000;
151 unsigned int timing_regAddress = base_timing_regAddress + 0x4*(counter); //0x80003C0 - 0x800003EC
152 unsigned int timing_regVal = 0x0; //the highest byte sets 8 (pdts trigger) + 1 (953.5 Hz) for burst mode, but 8 (pdts trigger) + 7 (single shot) for single
153 if (m_single_pulse) {
154 timing_regVal = 0xF0000000;
155 } else if (m_burst_mode) {
156 timing_regVal = 0x90000000;
157 }
158
159 TLOG(TLVL_FULL_DEBUG) << "Channel map is 0x" << std::hex << m_channel_mask << " and the comparison is 0x" << (1 << counter ) << std::endl;
160 if ( (m_channel_mask & ( (unsigned int)1 << counter)) == ((unsigned int)1 << counter) ) {
161 if ( counter < 6) {
162 bias_regVal = bias_regVal + pulse_bias_setting_270nm;
163 } else {
164 bias_regVal = bias_regVal + pulse_bias_setting_367nm;
165 }
166 timing_regVal = timing_regVal + m_pulse1_width_ticks;
167 timing_regVal = timing_regVal + (m_pulse2_width_ticks << 8);
168 timing_regVal = timing_regVal + (m_double_pulse_delay_ticks << 16);
169 TLOG(TLVL_FULL_DEBUG) << "Will turn on " << std::dec << counter << " channel at bias register 0x" << std::hex << bias_regAddress << " with bias value 0x" << bias_regVal << std::endl;
170 TLOG(TLVL_FULL_DEBUG) << " and set the width regsiter 0x" << std::hex << timing_regAddress << " to value of 0x" << timing_regVal << std::dec << std::endl;
171 m_device_interface->SetRegister(bias_regAddress, bias_regVal); //BIAS_DAC_CONFIG_N
172 if ( (counter == 7) && ! ( (m_channel_mask & ( (unsigned int)1 << counter)) == ((unsigned int)1 << (counter-1) ) ) ) {
173 //this is a really convoluted situation where for the very specific 12 channel board that arrived at CERN in June 2022
174 //the bias for channel 7 was not working and so the bias is taken from channel 6
175 //which means that if channel 7 is to be turned on while channel 6 is masked off
176 //you still have to bias channel 6 in order for the led on channel 7 to emmit light
177 m_device_interface->SetRegister((bias_regAddress - 0x4), bias_regVal); //BIAS_DAC_CONFIG_N
178 }
179
180 m_device_interface->SetRegister(timing_regAddress, timing_regVal); //cal_CONFIG_N
181 } else {
182 TLOG(TLVL_FULL_DEBUG) << "Will turn off channel " << std::dec << counter << " at timing register 0x" << std::hex << timing_regAddress << std::dec << std::endl;
183 m_device_interface->SetRegister(bias_regAddress, 0x0); //BIAS_DAC_CONFIG_N
184 m_device_interface->SetRegister(timing_regAddress, 0x0); //cal_CONFIG_N
185 }
186 }
187
188 m_device_interface->SetRegister(0x40000300, 0x1); //writing 0x1 to this register applies the bias voltage settings
189
190 m_run_marker = true;
191 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "Start pulsing SSPLEDCalibWrapper of card " << m_board_id << " complete.";
192}

194void
195SSPLEDCalibWrapper::stop(const data_t& /*args*/)
196{
197
198 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "Stop pulsing SSPLEDCalibWrapper of card " << m_board_id << "...";
199 if (!m_run_marker.load()) {
200 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "The run_marker says that SSPLEDCalibWrapper card " << m_board_id << " is already stopped, but stopping anyways...";
201 }
202
203 for (unsigned int counter = 0; counter < 12; counter++) { //switch this to 12 for a 12 channel SSP
204 unsigned int bias_regAddress = 0x40000340 + 0x4*(counter);
205 unsigned int timing_regAddress = 0x800003c0 + 0x4*(counter);
206 m_device_interface->SetRegister(bias_regAddress, 0x00000000); //BIAS_DAC_CONFIG_N
207 m_device_interface->SetRegister(timing_regAddress, 0x00000000); //cal_CONFIG_N
208 }
209
210 m_device_interface->SetRegister(0x40000300, 0x1); //writing 0x1 to this register applies the bias voltage settings
211
212 m_run_marker = false;
213 TLOG_DEBUG(TLVL_ENTER_EXIT_METHODS) << "Stop pulsing SSPLEDCalibWrapper of card " << m_board_id << " complete.";
214}
